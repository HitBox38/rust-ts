// Scratch-only stand-ins for render.rs items (straight ports) so source_map/styled_buffer can run.
class LineAnnotation {
  constructor(public start: any, public end: any, public kind: any, public label: any, public annotation_type: any, public highlight_source: boolean) {}
}
const unit = ($: string) => Object.freeze({ $ });
const ElementStyle: any = {
  MainHeaderMsg: unit("MainHeaderMsg"), HeaderMsg: unit("HeaderMsg"), LineAndColumn: unit("LineAndColumn"), LineNumber: unit("LineNumber"),
  Quotation: unit("Quotation"), UnderlinePrimary: unit("UnderlinePrimary"), UnderlineSecondary: unit("UnderlineSecondary"),
  LabelPrimary: unit("LabelPrimary"), LabelSecondary: unit("LabelSecondary"), NoStyle: unit("NoStyle"), Addition: unit("Addition"), Removal: unit("Removal"),
  color_spec(self: any, level: any, ss: any) {
    switch (self.$) {
      case "Addition": return ss.addition; case "Removal": return ss.removal; case "LineAndColumn": return ss.none; case "LineNumber": return ss.line_num;
      case "Quotation": return ss.none; case "MainHeaderMsg": return ss.emphasis; case "UnderlinePrimary": case "LabelPrimary": return level.style(ss);
      case "UnderlineSecondary": case "LabelSecondary": return ss.context; case "HeaderMsg": case "NoStyle": return ss.none; case "Level": throw new Error("x");
    }
  },
};
const CTRL = new Set([0,1,2,3,4,5,6,7,8,0xb,0xc,0xd,0xe,0xf,0x10,0x11,0x12,0x13,0x14,0x15,0x16,0x17,0x18,0x19,0x1a,0x1b,0x1c,0x1d,0x1e,0x1f,0x7f,0x202a,0x202b,0x202d,0x202e,0x2066,0x2067,0x2068,0x202c,0x2069]);
const debug_Loc = (v: any, f: any) => debug_struct(f, "Loc", [["line", String(v.line)], ["char", String(v.char)], ["display", String(v.display)], ["byte", String(v.byte)]]);
const debug_LAT = (v: any, f: any) => (v.$ === "Singleline" ? "Singleline" : debug_tuple(f, v.$, [String(v._0)]));

  num_overlap: (a_start: number, a_end: number, b_start: number, b_end: number, inclusive: boolean) => {

test("mirror", async () => {
  const sm_ = await import("/Users/tomernorman/Documents/personal/rust-ts/.data/lanes/annotate-a/src/vendor/annotate_snippets/renderer/source_map.ts");
  const { Margin, debug_Margin } = await import("/Users/tomernorman/Documents/personal/rust-ts/.data/lanes/annotate-a/src/vendor/annotate_snippets/renderer/margin.ts");
  const { StyledBuffer, debug_StyledBuffer } = await import("/Users/tomernorman/Documents/personal/rust-ts/.data/lanes/annotate-a/src/vendor/annotate_snippets/renderer/styled_buffer.ts");
  const { AnnotationKind, Annotation, Patch, debug_TrimmedPatch_ } = await import("/Users/tomernorman/Documents/personal/rust-ts/.data/lanes/annotate-a/src/vendor/annotate_snippets/snippet.ts") as any;

  const unhex = (s: string) => (s === "-" ? "" : Buffer.from(s, "hex").toString("utf8"));
  const pm = (e: unknown) => { if (e instanceof Panic) return e.message; throw e; };
  const style = (n: string) => ElementStyle[n];
  const tup = (xs: string[]) => debug_tuple(PLAIN, "", xs);
  const out: string[] = [];
  let source = Str.from(""), line_start = 1; let anns: any[] = []; let patches: any[] = []; let spans: [number, number][] = [];
  let sb = StyledBuffer.new();
  const dbgSH = (v: any, f: any) => sm_.debug_SubstitutionHighlight(v, f);
  for (const l of readFileSync("/Users/tomernorman/Documents/personal/rust-ts/.data/tmp/annotate-a/cases.txt", "utf8").split("\n")) {
    if (l === "") continue;
    const f = l.split(" ");

      case "source": source = Str.from(unhex(f[1]!)); line_start = +f[2]!; anns = []; patches = []; spans = []; break;

        const kind = f[1] === "P" ? "Primary" : f[1] === "C" ? "Context" : "Visible";

          out.push(`span ${a} ${b} ${s}`);
        }
        let s: string;

        out.push("annotated " + s);
        try {

      }
      case "substr": {

      case "puts": sb.puts(+f[1]!, +f[2]!, unhex(f[3]!), style(f[4]!)); break;
      case "append": sb.append(+f[1]!, unhex(f[2]!), style(f[3]!)); break;
      case "prepend": sb.prepend(+f[1]!, unhex(f[2]!), style(f[3]!)); break;
      case "replace": try { sb.replace(+f[1]!, +f[2]!, +f[3]!, unhex(f[4]!)); } catch (e) { out.push("sbpanic " + pm(e)); } break;

      case "sbdump": {

      }
